\begin{aligned}
N &=
1\cdot 2^0+
0\cdot 2^1+
1\cdot 2^2+
1\cdot 2^3+
0\cdot 2^4\\[4pt]
&=1+0+4+8+0\\[4pt]
&=\boxed{13}
\end{aligned}
\]
